cmake_minimum_required(VERSION 3.20)
project(cloudapi2_node LANGUAGES C CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|amd64|AMD64)$")
    message(FATAL_ERROR "This binding targets official Linux x86_64 native dependencies")
endif()
set(CLOUDAPI2_STAGED_SDK "" CACHE PATH "Private copied SDK source; never the original checkout")
set(NODE_INCLUDE_DIR "" CACHE PATH "Directory containing Node-API headers")
set(CLOUDAPI2_NATIVE_OUTPUT "${CMAKE_SOURCE_DIR}/native/linux-x64" CACHE PATH "Native output directory")
if(NOT EXISTS "${CLOUDAPI2_STAGED_SDK}/CMakeLists.txt" OR NOT EXISTS "${CLOUDAPI2_STAGED_SDK}/.cloudapi2-node-stage")
    message(FATAL_ERROR "Run scripts/build-native.mjs to stage a private SDK source copy first")
endif()
if(NOT EXISTS "${NODE_INCLUDE_DIR}/node_api.h")
    message(FATAL_ERROR "Set NODE_INCLUDE_DIR to the matching installed Node header directory")
endif()
set(ENABLE_JAVA OFF CACHE BOOL "No JNI/SWIG/JDK" FORCE)
add_subdirectory("${CLOUDAPI2_STAGED_SDK}" "${CMAKE_BINARY_DIR}/upstream")
file(STRINGS "${CLOUDAPI2_STAGED_SDK}/VERSION" CLOUDAPI2_UPSTREAM_VERSION LIMIT_COUNT 1)
add_library(cloudapi2_node MODULE native/addon.cpp)
option(CLOUDAPI2_CONTRACT_PROBES "Build isolated test-only native interfaces" OFF)
option(CLOUDAPI2_SANITIZERS "Instrument the Node binding for private ASan/UBSan tests" OFF)
if(CLOUDAPI2_SANITIZERS)
    target_compile_options(cloudapi2_node PRIVATE -O0 -g1 -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(cloudapi2_node PRIVATE -fsanitize=address,undefined)
endif()
if(CLOUDAPI2_CONTRACT_PROBES)
    target_compile_definitions(cloudapi2_node PRIVATE CLOUDAPI2_CONTRACT_PROBES=1)
    target_compile_options(cloudapi2_node PRIVATE -O0)
endif()
target_compile_features(cloudapi2_node PRIVATE cxx_std_20)
target_compile_definitions(cloudapi2_node PRIVATE NAPI_VERSION=8 CLOUDAPI2_UPSTREAM_VERSION="${CLOUDAPI2_UPSTREAM_VERSION}")
target_include_directories(cloudapi2_node PRIVATE "${NODE_INCLUDE_DIR}" "${CLOUDAPI2_STAGED_SDK}/src")
target_link_libraries(cloudapi2_node PRIVATE cloud_api2::flight_hub_sdk)
# Vendor libdrtc has an indirect libdrpc dependency and no usable bundled RUNPATH.
# DT_RPATH applies $ORIGIN to that indirect lookup without modifying vendor binaries.
target_link_options(cloudapi2_node PRIVATE "LINKER:--disable-new-dtags")
set_target_properties(cloudapi2_node PROPERTIES
    PREFIX "" SUFFIX ".node" CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES
    BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "$ORIGIN" INSTALL_RPATH_USE_LINK_PATH FALSE
    LIBRARY_OUTPUT_DIRECTORY "${CLOUDAPI2_NATIVE_OUTPUT}")
# Preserve dependency SONAME filenames. ldd/readelf validation is still required on Linux.
foreach(dep cloud_api2_crypto cloud_api2_ssl cloud_api2_websockets cloud_api2_paho cloud_api2_drpc cloud_api2_drtc_sdk)
    if(TARGET ${dep})
        add_custom_command(TARGET cloudapi2_node POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
            "$<TARGET_FILE:${dep}>" "$<TARGET_FILE_DIR:cloudapi2_node>/$<TARGET_FILE_NAME:${dep}>")
    endif()
endforeach()
